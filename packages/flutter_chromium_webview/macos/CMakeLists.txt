cmake_minimum_required(VERSION 3.19)
project(chromium_webview_macos LANGUAGES C CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "The macOS CEF wrapper and helpers require Xcode on macOS")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET 13.0)
set(CEF_TARGET_SDK 13.0 CACHE STRING "CEF deployment target" FORCE)
if(NOT CEF_ROOT)
  message(FATAL_ERROR "Run scripts/prepare_cef.py first")
endif()
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
set(USE_SANDBOX ON CACHE BOOL "CEF helper sandbox" FORCE)
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
set_target_properties(libcef_dll_wrapper PROPERTIES
  PREFIX "" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
add_executable(chromium_webview_helper Helpers/main.cc)
target_include_directories(chromium_webview_helper PRIVATE "${CEF_ROOT}")
target_compile_definitions(chromium_webview_helper PRIVATE CEF_USE_SANDBOX)
target_link_libraries(chromium_webview_helper PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
set_target_properties(chromium_webview_helper PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
